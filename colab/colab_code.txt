USER, PASS = "帳號", "密碼"   # 要跟 .env 的 VLM_USER / VLM_PASS 一樣

on_http_request:
  - actions:
      - type: basic-auth
        config:
          credentials:
            - "{USER}:{PASS}"
""")

import time, json, urllib.request

print("正在等待 ngrok 啟動並建立連線...")
for i in range(20):
    try:
        t = json.load(urllib.request.urlopen("http://localhost:4040/api/tunnels"))["tunnels"]
        if t:
            print("網址：", t[0]["public_url"])
            break
    except Exception:
        pass
    time.sleep(1)
else:
    print("失敗，log 如下：")
    print(open("ngrok.log").read()[-2000:])

# 測試
import base64, json, urllib.request
from PIL import Image, ImageDraw
img = Image.new("RGB", (640, 120), "navy")
ImageDraw.Draw(img).text((20, 40), "TEST 12345", fill="white")
img.save("t.jpg")
b64 = base64.b64encode(open("t.jpg", "rb").read()).decode()
req = {"model": "gemma4:12b", "stream": False, "think": False,
       "messages": [{"role": "user", "content": "圖中寫了什麼字？", "images": [b64]}]}
r = urllib.request.urlopen(urllib.request.Request(
    "http://localhost:11434/api/chat", json.dumps(req).encode(),
    {"Content-Type": "application/json"}))
print(json.load(r)["message"]["content"])
